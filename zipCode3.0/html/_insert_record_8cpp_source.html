92 ifstream newRecords(insertFile);
93 string newRec;
94 while (getline(newRecords, newRec)) {
95 string newKey = extractKey(newRec);
96 auto it = index.lower_bound(newKey);
97 int targetRBN = (it != index.end()) ? it->second : (--index.end())->second;
98
99 // Load the target block and insert new record
100 Block blk;
101 BlockBuffer::readBlock(static_cast<istream&>(data), blk, targetRBN, blockSize);
102 blk.records.push_back(newRec);
103
104 // Keep block sorted
105 sort(blk.records.begin(), blk.records.end(), [](const string& a, const string& b) {
106 return extractKey(a) < extractKey(b);
107 });
108
109 // Estimate block size
110 stringstream sizeSim;
111 for (auto& r : blk.records) sizeSim << r << '\n';
112
113 if (static_cast<int>(sizeSim.str().size()) < blockSize) {
114 // Block fits, update it
115 blk.recordCount = blk.records.size();
116 BlockBuffer::writeBlock(data, blk, targetRBN, blockSize);
117 } else {
118 // Split the block into two
119 vector<string> left(blk.records.begin(), blk.records.begin() + blk.records.size() / 2);
120 vector<string> right(blk.records.begin() + blk.records.size() / 2, blk.records.end());
121
122 Block leftBlock = { (int)left.size(), blk.prevRBN, nextRBN, left };
123 Block rightBlock = { (int)right.size(), targetRBN, blk.nextRBN, right };
124
125 BlockBuffer::writeBlock(data, leftBlock, targetRBN, blockSize);
126 BlockBuffer::writeBlock(data, rightBlock, nextRBN, blockSize);
127
128 // Update index with new keys

130 string highRight = extractKey(right.back());
131 index.erase(highLeft);
132 index[highLeft] = targetRBN;
133 index[highRight] = nextRBN;

138 }
